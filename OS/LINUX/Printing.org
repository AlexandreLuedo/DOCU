#+title: Printing
#+LANGUAGE: en
* Module docs
#+begin_src bash
man lpr
man lp
man lpstat
man lpadmin
#+end_src
* Print
#+begin_src bash
lpr /path/to/file
cat [file] | lpr
#+end_src
** Print black & white
#+begin_src bash
lpr -o color-mode=monochrome <file>
#+end_src
** Print specified pages
#+begin_src bash
lpr -o page-ranges=1 [file]
lpr -o page-ranges=1-3 [file]
#+end_src
* View printing queue
#+begin_src bash
lpq
#+end_src
* Cancel printing job
#+begin_src bash
lprm -
lprm [jobID]
lprm -a all
cancel [printerName]
#+end_src
* Setup lpr
#+begin_src bash
sudo lpadmin -d [printerName]
#+end_src
