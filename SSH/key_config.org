* For github
#+begin_src bash
[ -f ~/.ssh/id_ed25519 ] || ssh-keygen -t ed25519 -C "tonemail@example.com" -f ~/.ssh/id_ed25519 -N ""

echo -e "\nHost github.com\n  HostName github.com\n  User git\n  IdentityFile ~/.ssh/id_ed25519\n  IdentitiesOnly yes" >> ~/.ssh/config

# Run ssh agent, add key
eval "$(ssh-agent -s)"
ssh-add ~/.ssh/id_ed25519

# Affiche la clé publique à coller sur GitHub
cat ~/.ssh/id_ed25519.pub

# Teste la connexion SSH à GitHub
ssh -T git@github.com

#+end_src
* Typical usage of GPG keys
For github it's used to verify the commits
#+begin_src org
# Typical usage
gpg --encrypt file.txt             # Encrypt a file
gpg --sign file.txt                # Sign a file
gpg --verify file.txt.sig          # Verify someone's signature
gpg --list-keys                    # See your keys
#+end_src
* Create a full gpg key
See also: [[https://docs.github.com/en/authentication/managing-commit-signature-verification/generating-a-new-gpg-key][Github docs]]
#+begin_src bash :eval no
gpg --full-generate-key
# parameters => ECC (sign & encrypt) => Curve 25519 => 6m
#+end_src
