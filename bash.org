#+title: Bash/sh scripting
#+subtitle: Shell Scripting Cheat Sheet (sh/bash)
See also : [[https://devhints.io/bash]]
* Basic Structure
#+BEGIN_SRC bash
#!/bin/sh          # Shebang: #!/bin/bash or #!/usr/bin/env bash for portability
# Comments start with #

echo "Hello World" # Print to stdout
exit 0             # Exit with status code 0 (success)
#+END_SRC

* Variables
#+BEGIN_SRC bash
# Definition (no spaces around '=')
name="Lumo"
age=25

# Usage ($ or ${})
echo "Hello $name"
echo "Age: ${age} years"

# Special Variables
$0  # Script name
$1  # First argument
$@  # All arguments
$#  # Number of arguments
$?  # Exit status of last command
$$  # Process ID (PID)
#+END_SRC

* Conditionals (Tests)
#+BEGIN_SRC bash
if [ condition ]; then
    echo "True"
elif [ another_condition ]; then
    echo "Else if"
else
    echo "False"
fi

# File Tests
[ -f file ]    # Is a regular file
[ -d dir ]     # Is a directory
[ -r file ]    # Is readable
[ -x file ]    # Is executable
[ -z "$var" ]  # String is empty
[ -n "$var" ]  # String is not empty

# Numeric Comparisons
[ $a -eq $b ]  # Equal
[ $a -ne $b ]  # Not equal
[ $a -gt $b ]  # Greater than
[ $a -lt $b ]  # Less than
[ $a -ge $b ]  # Greater or equal
[ $a -le $b ]  # Less or equal

# String Comparisons
[ "$a" = "$b" ] # Equal
[ "$a" != "$b" ] # Not equal

# Logical Operators
[ cond1 ] && [ cond2 ]  # AND
[ cond1 ] || [ cond2 ]  # OR
! [ cond ]              # NOT
#+END_SRC

* Loops
#+BEGIN_SRC bash
# For Loop (List)
for i in 1 2 3; do
    echo $i
done

# For Loop (Globbing)
for file in *.txt; do
    echo "Processing: $file"
done

# While Loop
count=0
while [ $count -lt 5 ]; do
    echo $count
    count=$((count + 1))
done

# Until Loop (Run until true)
until [ -f lockfile ]; do
    sleep 1
done
#+END_SRC

* Functions
#+BEGIN_SRC bash
my_function() {
    local param=$1      # 'local' restricts scope
    echo "Param: $param"
    return 0            # Return status code
}

# Call function
my_function "test"
#+END_SRC

* Input / Output
#+BEGIN_SRC bash
# Read input
read -p "Enter name: " name
echo "You are $name"

# Redirections
cmd > file.txt      # Overwrite file
cmd >> file.txt     # Append to file
cmd 2> err.log      # Redirect stderr
cmd > file 2>&1     # Redirect stdout and stderr to same file

# Pipes
ls -l | grep ".txt" # Pipe stdout to next command
#+END_SRC

* Arithmetic
#+BEGIN_SRC bash
# POSIX arithmetic expansion
result=$(( 5 + 3 ))
echo $result

# Increment
i=$(( i + 1 ))
#+END_SRC

* File & Common Commands
#+BEGIN_SRC bash
cp src dst          # Copy
mv src dst          # Move/Rename
rm file             # Remove
mkdir dir           # Create directory
rmdir dir           # Remove empty directory
chmod +x script.sh  # Make executable
chown user:group f  # Change owner
touch file          # Create empty file/update timestamp
head -n 10 file     # First 10 lines
tail -f file        # Follow file (logs)
wc -l file          # Count lines
#+END_SRC

* Debugging & Safety Options
#+BEGIN_SRC bash
# Add these to the top of your script
set -e   # Exit immediately if a command exits with a non-zero status
set -u   # Treat unset variables as an error
set -x   # Print commands and their arguments as they are executed
#+END_SRC

* Tips
- Always quote variables: `"${variable}"` to handle spaces.
- Spaces are mandatory inside `[ condition ]`.
- Exit code `0` means success, anything else is an error.
- `#!/bin/sh` is portable; `#!/bin/bash` allows advanced features.
